---
title: Text Focus
description: A text element that focuses on hover.
category: text-animations
usage: |
  import TextFocus from "@/components/ui/text-focus";

  export function TextFocusDemo() {
    return (
      <TextFocus
        sentence="All Eyes on Me"
        borderColor="#22d3ee"
        blurAmount={3}
      />
    );
  }
props:
  - name: sentence
    type: string
    description: Sentence split into focusable words.
  - name: borderColor
    type: string
    default: '"#22d3ee"'
    description: Color of the animated focus frame.
  - name: blurAmount
    type: number
    default: "3"
    description: Blur applied to inactive words.
---
