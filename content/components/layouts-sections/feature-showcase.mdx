---
title: Feature Showcase
description: A feature section with animated visual states.
category: layouts-sections
usage: |
  import FeatureShowcase from "@/components/ui/feature-showcase";
  import { defaultFeatureItems } from "@/components/ui/feature-showcase";

  export function FeatureShowcaseDemo() {
    return (
      <FeatureShowcase
        items={defaultFeatureItems}
        title={"Easily deploy, maintain,\nand connect wallets"}
        subtitle="Launch embedded wallets in under 7 minutes."
      />
    );
  }
props:
  - name: items
    type: "FeatureShowcaseItem[]"
    description: Features displayed in the section.
  - name: title
    type: string
    description: Section heading rendered above the feature list.
  - name: subtitle
    type: string
    description: Supporting copy under the heading.
  - name: defaultActiveId
    type: number
    description: The initially active item id (defaults to the first item).
  - name: activeColor
    type: string
    default: '"text-blue-600"'
    description: Tailwind color class applied to the active accent.
  - name: className
    type: string
    description: Additional classes for the root element.
---
